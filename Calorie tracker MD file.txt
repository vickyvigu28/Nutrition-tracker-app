# Nutrition Tracker App - Phase 2 Build Specification (GitHub Sync)

## Project Overview
Build a mobile-first nutrition tracking app with:
- User profile setup (auto-calculate macros based on BMI)
- Food logging with OpenAI parsing
- Real-time dashboard with macro/micro tracking
- Weekly trends visualization
- **GitHub sync** (offline-first, manual sync)
- Side drawer navigation with centered title

**Tech Stack**: React 18, Vite, Tailwind CSS, Chart.js, GitHub API, OpenAI API

## 5. GitHub Sync Service

**File**: `src/services/githubService.js`

```javascript
import axios from 'axios';
import { Base64 } from 'js-base64';

const GITHUB_API = 'https://api.github.com';
const GITHUB_TOKEN = import.meta.env.VITE_GITHUB_TOKEN;
const GITHUB_OWNER = import.meta.env.VITE_GITHUB_OWNER;
const GITHUB_REPO = import.meta.env.VITE_GITHUB_REPO;
const GITHUB_BRANCH = import.meta.env.VITE_GITHUB_BRANCH;

const headers = {
  'Authorization': `token ${GITHUB_TOKEN}`,
  'Accept': 'application/vnd.github.v3+json'
};

export const downloadUserData = async (userId) => {
  try {
    const fileName = `${userId}.json`;
    const path = `data/${fileName}`;
    
    const response = await axios.get(
      `${GITHUB_API}/repos/${GITHUB_OWNER}/${GITHUB_REPO}/contents/${path}?ref=${GITHUB_BRANCH}`,
      { headers }
    );
    
    const decodedContent = Base64.decode(response.data.content);
    const userData = JSON.parse(decodedContent);
    
    return {
      data: userData,
      sha: response.data.sha,
      exists: true
    };
  } catch (error) {
    if (error.response?.status === 404) {
      return { data: null, sha: null, exists: false };
    }
    throw new Error(`Failed to download: ${error.message}`);
  }
};

export const uploadUserData = async (userId, userData, sha = null) => {
  try {
    const fileName = `${userId}.json`;
    const path = `data/${fileName}`;
    
    const content = Base64.encode(JSON.stringify(userData, null, 2));
    
    const payload = {
      message: `Update nutrition data for ${userId} - ${new Date().toISOString()}`,
      content: content,
      branch: GITHUB_BRANCH,
      ...(sha && { sha })
    };
    
    const response = await axios.put(
      `${GITHUB_API}/repos/${GITHUB_OWNER}/${GITHUB_REPO}/contents/${path}`,
      payload,
      { headers }
    );
    
    return {
      success: true,
      sha: response.data.content.sha,
      timestamp: new Date().toISOString()
    };
  } catch (error) {
    throw new Error(`Failed to upload: ${error.message}`);
  }
};

export const syncData = async (userId, localData) => {
  try {
    const remote = await downloadUserData(userId);
    
    if (!remote.exists) {
      console.log('First sync: uploading local data');
      const result = await uploadUserData(userId, localData);
      return { data: localData, synced: true, result };
    }
    
    const remoteData = remote.data;
    const localTime = new Date(localData.last_synced || 0);
    const remoteTime = new Date(remoteData.last_synced || 0);
    
    if (localTime > remoteTime) {
      console.log('Local data is newer: uploading');
      const result = await uploadUserData(userId, localData, remote.sha);
      return { data: localData, synced: true, result };
    }
    
    console.log('Remote data is newer: merging');
    const merged = mergeData(localData, remoteData);
    await uploadUserData(userId, merged, remote.sha);
    return { data: merged, synced: true, result: 'merged' };
  } catch (error) {
    console.error('Sync error:', error);
    throw error;
  }
};

const mergeData = (localData, remoteData) => {
  const localMealIds = new Set(localData.meals.map(m => m.id));
  const remoteMealIds = new Set(remoteData.meals.map(m => m.id));
  
  const newRemoteMeals = remoteData.meals.filter(m => !localMealIds.has(m.id));
  const mergedMeals = [...localData.meals, ...newRemoteMeals];
  
  const mergedDailyTotals = {
    ...remoteData.daily_totals,
    ...localData.daily_totals
  };
  
  return {
    ...localData,
    meals: mergedMeals,
    daily_totals: mergedDailyTotals,
    last_synced: new Date().toISOString()
  };
};

export const getSyncHistory = async (userId) => {
  try {
    const fileName = `${userId}.json`;
    const path = `data/${fileName}`;
    
    const response = await axios.get(
      `${GITHUB_API}/repos/${GITHUB_OWNER}/${GITHUB_REPO}/commits?path=${path}&per_page=10`,
      { headers }
    );
    
    return response.data.map(commit => ({
      date: commit.commit.author.date,
      message: commit.commit.message,
      sha: commit.sha
    }));
  } catch (error) {
    console.error('Failed to get history:', error);
    return [];
  }
};

export const testConnection = async () => {
  try {
    await axios.get(
      `${GITHUB_API}/repos/${GITHUB_OWNER}/${GITHUB_REPO}`,
      { headers }
    );
    return { connected: true };
  } catch (error) {
    return { connected: false, error: error.message };
  }
};
```

## 6. BMI & Macro Calculation Logic

**File**: `src/utils/bmiCalculator.js`

```javascript
export const calculateBMI = (heightCm, weightKg) => {
  const heightM = heightCm / 100;
  return parseFloat((weightKg / (heightM * heightM)).toFixed(1));
};

export const calculateMacros = (heightCm, weightKg, goalWeightKg, age, gender, goals) => {
  const bmi = calculateBMI(heightCm, weightKg);
  
  let bmr;
  if (gender === 'male') {
    bmr = 88.362 + (13.397 * weightKg) + (4.799 * heightCm) - (5.677 * age);
  } else {
    bmr = 447.593 + (9.247 * weightKg) + (3.098 * heightCm) - (4.330 * age);
  }
  
  const tdee = bmr * 1.55;
  
  let calories = tdee;
  if (weightKg > goalWeightKg) {
    calories = tdee - 500;
  }
  
  const proteinG = weightKg * 1.8;
  const fatG = (calories * 0.25) / 9;
  let carbsG = (calories - (proteinG * 4) - (fatG * 9)) / 4;
  let fiberG = weightKg * 0.4;
  
  let calciumMg = 1000;
  let zincMg = gender === 'male' ? 11 : 8;
  
  if (goals.includes('lower_ldl')) {
    fiberG = Math.max(fiberG, 35);
  }
  
  return {
    bmi,
    tdee: Math.round(tdee),
    calories_daily: Math.round(calories),
    protein_g: Math.round(proteinG),
    carbs_g: Math.round(carbsG),
    fat_g: Math.round(fatG),
    fiber_g: Math.round(fiberG),
    calcium_mg: calciumMg,
    zinc_mg: zincMg
  };
};
```

## 7. OpenAI Integration for Food Parsing

**File**: `src/services/openaiService.js`

```javascript
const OPENAI_API_KEY = import.meta.env.VITE_OPENAI_API_KEY;

export const parseFoodInput = async (userInput, clarifications = []) => {
  const systemPrompt = `You are a nutrition data validator, not a guesser.
Your job is to:
1. Parse food items mentioned
2. Extract quantities stated
3. Identify ambiguities that need clarification
4. Suggest clarifying questions for Indian cuisine

Do NOT estimate calories or macros. Your job is to clarify intent only.

Return ONLY valid JSON with no markdown:
{
  "parsed_items": [
    {
      "item": "egg",
      "quantity": 1,
      "confidence": "high"
    }
  ],
  "ambiguities": [
    {
      "item": "egg",
      "question": "What size?",
      "options": ["small", "medium", "large"],
      "hint": "Medium is ~50g"
    }
  ]
}`;

  const userMessage = clarifications.length > 0 
    ? `Previous clarifications: ${JSON.stringify(clarifications)}\n\nNow confirm these items and return their nutrition as JSON array.`
    : `Parse this food input: "${userInput}"`;

  try {
    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${OPENAI_API_KEY}`
      },
      body: JSON.stringify({
        model: 'gpt-4o-mini',
        messages: [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: userMessage }
        ],
        temperature: 0.7,
        max_tokens: 500
      })
    });

    const data = await response.json();
    if (!response.ok) {
      throw new Error(data.error?.message || 'OpenAI API error');
    }
    
    const content = data.choices[0].message.content;
    const jsonMatch = content.match(/\{[\s\S]*\}|\[[\s\S]*\]/);
    return JSON.parse(jsonMatch[0]);
  } catch (error) {
    console.error('OpenAI API error:', error);
    throw error;
  }
};

export const getNutritionData = async (foodItems) => {
  const systemPrompt = `You are a nutrition lookup service. Given confirmed food items with portions, return accurate nutrition data.
Use USDA FoodData Central estimates. Be conservative - don't inflate numbers.

Return ONLY valid JSON array with no markdown:
[
  {
    "food_name": "egg, medium, boiled",
    "portion": "1 medium (50g)",
    "calories": 78,
    "protein_g": 6.3,
    "carbs_g": 0.6,
    "fat_g": 5.3,
    "fiber_g": 0,
    "calcium_mg": 28,
    "zinc_mg": 0.6,
    "source": "USDA"
  }
]`;

  try {
    const response = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${OPENAI_API_KEY}`
      },
      body: JSON.stringify({
        model: 'gpt-4o-mini',
        messages: [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: `Get nutrition for: ${JSON.stringify(foodItems)}` }
        ],
        temperature: 0.5,
        max_tokens: 800
      })
    });

    const data = await response.json();
    if (!response.ok) {
      throw new Error(data.error?.message || 'OpenAI API error');
    }
    
    const content = data.choices[0].message.content;
    const jsonMatch = content.match(/\[[\s\S]*\]/);
    return JSON.parse(jsonMatch[0]);
  } catch (error) {
    console.error('Nutrition lookup error:', error);
    throw error;
  }
};
```

```jsx
import React, { useState, useEffect } from 'react';

export default function Dashboard({ todayData, targets }) {
  const [showMicros, setShowMicros] = useState(false);

  const macros = [
    { key: 'protein_g', label: 'Protein', color: 'bg-blue-100 text-blue-700' },
    { key: 'carbs_g', label: 'Carbs', color: 'bg-amber-100 text-amber-700' },
    { key: 'fiber_g', label: 'Fiber', color: 'bg-green-100 text-green-700' },
    { key: 'fat_g', label: 'Fat', color: 'bg-red-100 text-red-700' }
  ];

  const micros = [
    { label: 'Calcium', current: todayData?.calcium_mg, target: targets.calcium_mg, unit: 'mg' },
    { label: 'Zinc', current: todayData?.zinc_mg, target: targets.zinc_mg, unit: 'mg' }
  ];

  const getStatus = (current, target) => {
    if (!current || !target) return 'neutral';
    const percent = (current / target) * 100;
    if (percent >= 95 && percent <= 105) return 'success';
    if (percent > 105) return 'warning';
    return 'info';
  };

  return (
    <div className="space-y-4 p-4">
      <div className="bg-white rounded-lg p-4 border border-gray-200">
        <div className="text-sm text-gray-600 mb-2">Total calories</div>

        </div>
      </div>

      <div className="grid grid-cols-2 gap-3">
        {macros.map(macro => (
```

### **Settings.jsx**

```jsx
import React, { useState, useEffect } from 'react';
import { calculateMacros } from '../utils/bmiCalculator';

export default function Settings({ profile, targets, onSave }) {
  const [formData, setFormData] = useState(profile);
  const [calculatedTargets, setCalculatedTargets] = useState(targets);
  const [overrides, setOverrides] = useState({});

  useEffect(() => {
    const calculated = calculateMacros(
      formData.height_cm,
      formData.weight_kg,
      formData.goal_weight_kg,
      formData.age,
      formData.gender,
      formData.goals || []
    );
    setCalculatedTargets(calculated);
  }, [formData]);

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: name.includes('cm') || name.includes('kg') || name === 'age' ? parseFloat(value) : value }));
  };

  const handleOverride = (macro, value) => {
    setOverrides(prev => ({ ...prev, [macro]: parseInt(value) || 0 }));
  };

  const macroFields = [
    { key: 'calories_daily', label: 'Daily calories', unit: 'kcal' },
    { key: 'protein_g', label: 'Protein', unit: 'g' },
    { key: 'carbs_g', label: 'Carbs', unit: 'g' },
    { key: 'fat_g', label: 'Fat', unit: 'g' },
    { key: 'fiber_g', label: 'Fiber', unit: 'g' }
  ];

  return (
    <div className="p-4 space-y-6">
      <h2 className="text-lg font-bold">Profile Settings</h2>

      <div>
        <h3 className="font-semibold text-sm mb-3">Basic Information</h3>
        
        <div className="space-y-3">
          <div>
            <label className="block text-sm font-medium mb-2">Name</label>
            <input 
              type="text" 
              name="name"
              value={formData.name}
              onChange={handleInputChange}
              className="w-full p-2 border rounded-lg"
            />
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium mb-2">Height (cm)</label>
              <input 
                type="number"
                name="height_cm"
                value={formData.height_cm}
                onChange={handleInputChange}
                className="w-full p-2 border rounded-lg"
              />
            </div>
            <div>
              <label className="block text-sm font-medium mb-2">Weight (kg)</label>
              <input 
                type="number"
                name="weight_kg"
                value={formData.weight_kg}
                onChange={handleInputChange}
                className="w-full p-2 border rounded-lg"
              />
            </div>
          </div>

          <div>
            <label className="block text-sm font-medium mb-2">Goal Weight (kg)</label>
            <input 
              type="number"
              name="goal_weight_kg"
              value={formData.goal_weight_kg}
              onChange={handleInputChange}
              className="w-full p-2 border rounded-lg"
            />
          </div>
        </div>
      </div>

      <div>
        <h3 className="font-semibold text-sm mb-3">Nutrition Targets (Auto-calculated - override if needed)</h3>
        
        <div className="bg-blue-50 border border-blue-200 rounded-lg p-4 space-y-3">
          {macroFields.map(field => (
            <div key={field.key} className="flex items-center gap-3">
              <label className="text-sm font-medium flex-1">{field.label}</label>
              <input 
                type="number"
                defaultValue={calculatedTargets[field.key]}
                onChange={(e) => handleOverride(field.key, e.target.value)}
                className="w-20 p-2 border rounded text-sm"
              />
              <span className="text-sm text-gray-600 w-8">{field.unit}</span>
            </div>
          ))}
        </div>
      </div>

      <button 
        onClick={() => onSave({ profile: formData, targets: { ...calculatedTargets, ...overrides } })}
        className="w-full bg-blue-600 text-white py-3 rounded-lg font-semibold hover:bg-blue-700"
      >
        Save Profile
      </button>
    </div>
  );
}
```

## 14. GitHub Sync Features

### Advantages
- **Version history**: Every sync creates a commit
- **Rollback**: Easy to revert meal entries if needed
- **Multi-user**: Each person's file separate (no conflicts)
- **Audit trail**: See who changed what and when
- **Free backup**: Clone repo = full backup

### Workflow
User adds food locally
→ localStorage updated
→ User clicks "Sync"
→ App commits to GitHub
→ Sync indicator shows ✓
→ If offline: queue sync for later
